 function acc(b,title,icon,count,fill){
  const box=mk("div","gAcc");
  const h=mk("button","gHead",RBI.svg(icon)+'<span class="lb"></span><small></small>'+RBI.svg("chev"));
  h.querySelector(".lb").textContent=title;h.querySelector("small").textContent=count||"";
  const body=mk("div","gBody"),inner=mk("div","gIn");body.appendChild(inner);
  let filled=false;
  h.onclick=()=>{
   if(!filled){fill(inner);filled=true}
   const open=!box.classList.contains("open");
   if(open){body.style.maxHeight=inner.scrollHeight+"px";box.classList.add("open");setTimeout(()=>{if(box.classList.contains("open"))body.style.maxHeight="none"},300)}
   else{body.style.maxHeight=inner.scrollHeight+"px";void body.offsetWidth;box.classList.remove("open");body.style.maxHeight="0px"}
   buzz(6);
  };
  box.appendChild(h);box.appendChild(body);b.appendChild(box);return box;
 }
