/* ===== RoomBoom offline layer ====================================================
   - RBC: tiny localStorage cache (chat list, messages per chat, profiles, notifications)
   - outbox: text messages typed while offline are queued and sent when the net is back
   Loaded BEFORE the main script. Uses sb / me / msgs / draw / room etc. only at call time. */
const RBC=(function(){
 const K="rb_c_";
 function get(k){try{const v=localStorage.getItem(K+k);return v?JSON.parse(v):null}catch(e){return null}}
 function purge(){try{Object.keys(localStorage).filter(x=>x.startsWith(K+"msgs_")).forEach(x=>localStorage.removeItem(x))}catch(e){}}
 function set(k,v){
  try{localStorage.setItem(K+k,JSON.stringify(v));return true}
  catch(e){try{purge();localStorage.setItem(K+k,JSON.stringify(v));return true}catch(e2){return false}}
 }
 function del(k){try{localStorage.removeItem(K+k)}catch(e){}}
 function clearAll(){
  try{indexedDB.deleteDatabase("rb_media")}catch(e){}
  try{Object.keys(localStorage).filter(x=>x.startsWith(K)||x==="rb_outbox"||x==="rb_ann_seen"||x==="rb_pins"||x==="rb_mutes"||x==="rb_arch"||x==="rb_stars"||x==="rb_stats").forEach(x=>localStorage.removeItem(x))}catch(e){}
 }
 function outbox(){try{return(JSON.parse(localStorage.getItem("rb_outbox"))||[]).map(i=>{if(!i.id)i.id=i.tempId||("q-"+Math.random().toString(36).slice(2));return i})}catch(e){return []}}
 function saveOutbox(a){try{localStorage.setItem("rb_outbox",JSON.stringify(a))}catch(e){}}
 return {get,set,del,clearAll,outbox,saveOutbox};
})();


/* ---------- RBlog: a small persistent error log (survives restarts) so bugs can be reported with real details ---------- */
const RBlog=(()=>{
 const K="rb_errlog";
 const get=()=>{try{return JSON.parse(localStorage.getItem(K))||[]}catch(e){return[]}};
 const add=(kind,msg)=>{
  try{
   let m=String((msg&&msg.message)||msg||"").slice(0,240);
   if(!m||/ResizeObserver loop|^Script error\.?$/.test(m))return;
   const a=get(),last=a[a.length-1];
   if(last&&last.m===m&&Date.now()-last.t<3000)return;
   let v="";try{v=curView}catch(e){}
   a.push({t:Date.now(),k:kind,m,v});while(a.length>40)a.shift();
   localStorage.setItem(K,JSON.stringify(a));
  }catch(e){}
 };
 addEventListener("error",e=>add("js",(e.message||"error")+" @"+String(e.filename||"").split("/").pop()+":"+e.lineno));
 addEventListener("unhandledrejection",e=>add("promise",(e.reason&&e.reason.message)||e.reason));
 return{get,add,clear:()=>{try{localStorage.removeItem(K)}catch(e){}}};
})();

/* RBS: small local flags per chat (pinned / muted) */
const RBS={
 get(k){try{return new Set(JSON.parse(localStorage.getItem(k))||[])}catch(e){return new Set()}},
 save(k,s){try{localStorage.setItem(k,JSON.stringify([...s]))}catch(e){}},
 has(k,id){return RBS.get(k).has(id)}
};
window.__lpAt=0; /* time of the last long-press, so the tap that follows it is ignored */

/* is this error just "no internet"? */
function isNetErr(err){
 const m=((err&&err.message)||String(err||"")).toLowerCase();
 return !navigator.onLine||m.includes("fetch")||m.includes("network")||m.includes("failed to")||m.includes("load failed")||(err&&err.name==="TypeError");
}

/* ---------- messages cache (per chat, newest 60, text only is enough for reading offline) ---------- */
let _pmT=null;
function persistMsgs(){
 clearTimeout(_pmT);
 _pmT=setTimeout(()=>{
  if(!room)return;
  const arr=[...msgs.values()].filter(m=>!m._pending).sort((a,b)=>T(a.created_at)-T(b.created_at)).slice(-60);
  RBC.set("msgs_"+room,arr);
 },400);
}

/* ---------- outbox: every message (text, photo, voice) shows instantly and is sent in the background ---------- */
/* big files can't live in localStorage, so photo/voice blobs wait in IndexedDB (with an in-memory fallback) */
const RBM=(()=>{
 const mem=new Map();let dbp=null;
 const open=()=>dbp||(dbp=new Promise((res,rej)=>{try{const r=indexedDB.open("rb_media",1);r.onupgradeneeded=()=>r.result.createObjectStore("b");r.onsuccess=()=>res(r.result);r.onerror=()=>rej(r.error)}catch(e){rej(e)}}));
 const tx=async(mode,fn)=>{const d=await open();return new Promise((res,rej)=>{const t=d.transaction("b",mode),rq=fn(t.objectStore("b"));t.oncomplete=()=>res(rq&&rq.result);t.onerror=()=>rej(t.error)})};
 return{
  put:async(k,v)=>{mem.set(k,v);try{await tx("readwrite",s=>s.put(v,k))}catch(e){}},
  get:async k=>{if(mem.has(k))return mem.get(k);try{return await tx("readonly",s=>s.get(k))}catch(e){return null}},
  del:async k=>{mem.delete(k);try{await tx("readwrite",s=>s.delete(k))}catch(e){}}
 };
})();
const localURL=new Map(); /* temp preview of photos we are sending */

function queuedBubble(it){
 const r=it.row;
 return {id:it.id,room_id:r.room_id,sender_id:r.sender_id,body:r.body||null,media_type:r.media_type||null,_local:localURL.get(it.id)||null,
  created_at:it.at||new Date().toISOString(),reply_to:r.reply_to||null,_pending:true,_queued:!navigator.onLine||!!it.wait};
}
function queueItem(it){
 const q=RBC.outbox();q.push(it);RBC.saveOutbox(q);
}
function queueText(row){ /* shows the bubble right now, then sends */
 row.id=row.id||crypto.randomUUID();
 const it={id:row.id,row,at:new Date().toISOString()};
 queueItem(it);
 const b=queuedBubble(it);msgs.set(b.id,b);draw(b);
 flushOutbox();
}
function queuedFor(roomId){return RBC.outbox().filter(i=>i.row.room_id===roomId).map(queuedBubble)}
function setPendState(id,wait){
 const q=RBC.outbox(),it=q.find(x=>x.id===id);if(it&&!!it.wait!==wait){it.wait=wait;RBC.saveOutbox(q)}
 const b=msgs.get(id);if(b&&b._pending){b._queued=wait||!navigator.onLine;draw(b)}
}
function repaintPending(){msgs.forEach(b=>{if(b._pending){const q=!navigator.onLine||!!(RBC.outbox().find(x=>x.id===b.id)||{}).wait;if(b._queued!==q){b._queued=q;draw(b)}}})}

async function sendItem(it){
 const row=Object.assign({},it.row);
 if(it.media){
  const blob=await RBM.get(it.id);
  if(!blob)return{error:{message:"file is missing"}};
  const {error:upErr}=await sb.storage.from("media").upload(row.media_path,blob,{contentType:it.ctype||blob.type||"application/octet-stream"});
  if(upErr&&!/already exists|duplicate/i.test(upErr.message||""))return{error:upErr};
 }
 const {data,error}=await sb.from("messages").insert(row).select().single();
 if(error){
  if(row.id&&(error.code==="23505"||/duplicate key/i.test(error.message||""))){ /* a retry after a lost reply: it is already there */
   const r=await sb.from("messages").select("*").eq("id",row.id).single();if(r.data)return{data:r.data};
  }
  return{error};
 }
 return{data};
}

let _flushing=false,_again=false;
async function flushOutbox(){
 if(typeof me==="undefined"||!me)return;
 if(_flushing){_again=true;return}
 if(!navigator.onLine){repaintPending();return}
 _flushing=true;
 try{
  do{
   _again=false;
   for(;;){
    const q=RBC.outbox();if(!q.length)break;
    const it=q[0];
    setPendState(it.id,false);
    let res;try{res=await sendItem(it)}catch(e){res={error:e}}
    if(res.error){
     if(isNetErr(res.error)){setPendState(it.id,true);break}
     /* the server refused it for real: take the bubble away and say why */
     toast(res.error,"err");
     msgs.delete(it.id);const el=$("m-"+it.id);if(el)el.remove();
     if(!it.media&&it.row.body&&$("txt")&&!$("txt").value)$("txt").value=it.row.body;
    }else{
     const data=res.data;
     if(room===it.row.room_id){msgs.set(data.id,data);draw(data);persistMsgs()}else msgs.delete(it.id);
    }
    RBC.saveOutbox(RBC.outbox().filter(x=>x.id!==it.id));
    if(it.media){RBM.del(it.id);const u=localURL.get(it.id);if(u)setTimeout(()=>{URL.revokeObjectURL(u);localURL.delete(it.id)},4000)}
   }
  }while(_again);
 }finally{_flushing=false}
 if(typeof curView!=="undefined"&&curView==="rooms")loadRooms();
}
/* keep trying while something is waiting (covers "online but the server is not reachable") */
setInterval(()=>{if(navigator.onLine&&RBC.outbox().length)flushOutbox()},12000);
addEventListener("offline",()=>{try{repaintPending()}catch(e){}});
