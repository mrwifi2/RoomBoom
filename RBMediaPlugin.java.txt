package com.roomboom.app;

import android.Manifest;
import android.content.ContentResolver;
import android.content.ContentUris;
import android.database.Cursor;
import android.graphics.Bitmap;
import android.graphics.BitmapFactory;
import android.graphics.Matrix;
import android.net.Uri;
import android.os.Build;
import android.provider.MediaStore;
import android.util.Base64;
import android.util.Size;
import androidx.activity.result.ActivityResult;
import androidx.core.content.FileProvider;
import android.media.ExifInterface;
import com.getcapacitor.JSArray;
import com.getcapacitor.JSObject;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.ActivityCallback;
import com.getcapacitor.annotation.CapacitorPlugin;
import com.getcapacitor.annotation.Permission;
import com.getcapacitor.annotation.PermissionCallback;
import java.io.ByteArrayOutputStream;
import java.io.File;
import java.io.InputStream;

@CapacitorPlugin(name = "RBMedia", permissions = {
  @Permission(alias = "images", strings = { Manifest.permission.READ_MEDIA_IMAGES }),
  @Permission(alias = "legacy", strings = { Manifest.permission.READ_EXTERNAL_STORAGE })
})
public class RBMediaPlugin extends Plugin {
  private File camFile;

  private String alias() { return Build.VERSION.SDK_INT >= 33 ? "images" : "legacy"; }

  @PluginMethod
  public void list(PluginCall call) {
    if (getPermissionState(alias()) != com.getcapacitor.PermissionState.GRANTED) {
      requestPermissionForAlias(alias(), call, "listPerm");
      return;
    }
    doList(call);
  }

  @PermissionCallback
  private void listPerm(PluginCall call) {
    if (getPermissionState(alias()) == com.getcapacitor.PermissionState.GRANTED) doList(call);
    else call.reject("permission denied");
  }

  private void doList(final PluginCall call) {
    final int limit = call.getInt("limit", 60), offset = call.getInt("offset", 0);
    new Thread(() -> {
      try {
        JSArray out = new JSArray();
        ContentResolver cr = getContext().getContentResolver();
        Uri base = MediaStore.Images.Media.EXTERNAL_CONTENT_URI;
        String[] proj = { MediaStore.Images.Media._ID, MediaStore.Images.Media.DATE_ADDED };
        Cursor c;
        if (Build.VERSION.SDK_INT >= 26) {
          android.os.Bundle b = new android.os.Bundle();
          b.putInt(ContentResolver.QUERY_ARG_LIMIT, limit);
          b.putInt(ContentResolver.QUERY_ARG_OFFSET, offset);
          b.putStringArray(ContentResolver.QUERY_ARG_SORT_COLUMNS, new String[] { MediaStore.Images.Media.DATE_ADDED });
          b.putInt(ContentResolver.QUERY_ARG_SORT_DIRECTION, ContentResolver.QUERY_SORT_DIRECTION_DESCENDING);
          c = cr.query(base, proj, b, null);
        } else {
          c = cr.query(base, proj, null, null, "date_added DESC LIMIT " + limit + " OFFSET " + offset);
        }
        if (c != null) {
          while (c.moveToNext()) {
            long id = c.getLong(0);
            Uri u = ContentUris.withAppendedId(base, id);
            Bitmap bm = null;
            try {
              if (Build.VERSION.SDK_INT >= 29) bm = cr.loadThumbnail(u, new Size(220, 220), null);
              else bm = MediaStore.Images.Thumbnails.getThumbnail(cr, id, MediaStore.Images.Thumbnails.MINI_KIND, null);
            } catch (Exception e) { }
            if (bm == null) continue;
            JSObject o = new JSObject();
            o.put("id", String.valueOf(id));
            o.put("thumb", enc(bm, 70));
            out.put(o);
          }
          c.close();
        }
        JSObject r = new JSObject();
        r.put("items", out);
        call.resolve(r);
      } catch (Exception e) { call.reject(String.valueOf(e.getMessage())); }
    }).start();
  }

  @PluginMethod
  public void get(final PluginCall call) {
    final String id = call.getString("id");
    new Thread(() -> {
      try {
        Uri u = ContentUris.withAppendedId(MediaStore.Images.Media.EXTERNAL_CONTENT_URI, Long.parseLong(id));
        JSObject r = new JSObject();
        r.put("data", loadScaled(u));
        call.resolve(r);
      } catch (Exception e) { call.reject(String.valueOf(e.getMessage())); }
    }).start();
  }

  @PluginMethod
  public void camera(PluginCall call) {
    try {
      camFile = new File(getContext().getCacheDir(), "rb_cam_" + System.currentTimeMillis() + ".jpg");
      Uri u = FileProvider.getUriForFile(getContext(), getContext().getPackageName() + ".fileprovider", camFile);
      android.content.Intent i = new android.content.Intent(MediaStore.ACTION_IMAGE_CAPTURE);
      i.putExtra(MediaStore.EXTRA_OUTPUT, u);
      i.addFlags(android.content.Intent.FLAG_GRANT_WRITE_URI_PERMISSION | android.content.Intent.FLAG_GRANT_READ_URI_PERMISSION);
      startActivityForResult(call, i, "camDone");
    } catch (Exception e) { call.reject(String.valueOf(e.getMessage())); }
  }

  @ActivityCallback
  private void camDone(PluginCall call, ActivityResult res) {
    if (call == null) return;
    if (res.getResultCode() != android.app.Activity.RESULT_OK || camFile == null || !camFile.exists()) { call.reject("cancelled"); return; }
    try {
      JSObject r = new JSObject();
      r.put("data", loadScaled(Uri.fromFile(camFile)));
      camFile.delete();
      call.resolve(r);
    } catch (Exception e) { call.reject(String.valueOf(e.getMessage())); }
  }

  private String enc(Bitmap b, int q) {
    ByteArrayOutputStream bo = new ByteArrayOutputStream();
    b.compress(Bitmap.CompressFormat.JPEG, q, bo);
    return Base64.encodeToString(bo.toByteArray(), Base64.NO_WRAP);
  }

  private String loadScaled(Uri u) throws Exception {
    ContentResolver cr = getContext().getContentResolver();
    BitmapFactory.Options o = new BitmapFactory.Options();
    o.inJustDecodeBounds = true;
    InputStream in = cr.openInputStream(u);
    BitmapFactory.decodeStream(in, null, o);
    in.close();
    int s = 1;
    while (Math.max(o.outWidth, o.outHeight) / s > 2560) s *= 2;
    o = new BitmapFactory.Options();
    o.inSampleSize = s;
    in = cr.openInputStream(u);
    Bitmap bm = BitmapFactory.decodeStream(in, null, o);
    in.close();
    int rot = 0;
    try {
      in = cr.openInputStream(u);
      int ori = new ExifInterface(in).getAttributeInt(ExifInterface.TAG_ORIENTATION, 1);
      in.close();
      if (ori == 6) rot = 90; else if (ori == 3) rot = 180; else if (ori == 8) rot = 270;
    } catch (Exception e) { }
    float sc = Math.min(1f, 1280f / Math.max(bm.getWidth(), bm.getHeight()));
    Matrix m = new Matrix();
    m.postScale(sc, sc);
    if (rot != 0) m.postRotate(rot);
    Bitmap out = Bitmap.createBitmap(bm, 0, 0, bm.getWidth(), bm.getHeight(), m, true);
    return enc(out, 78);
  }
}
