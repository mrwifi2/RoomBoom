package com.roomboom.app;

import android.content.Intent;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.CapacitorPlugin;
import com.myterminal.app.TerminalActivity;

/** Opens MyTerminal (the built-in Linux terminal) as a screen inside this same app. */
@CapacitorPlugin(name = "RBTerm")
public class RBTermPlugin extends Plugin {
  @PluginMethod
  public void open(PluginCall call) {
    try {
      Intent i = new Intent(getContext(), TerminalActivity.class);
      getActivity().startActivity(i);
      call.resolve();
    } catch (Exception e) {
      call.reject(String.valueOf(e.getMessage()));
    }
  }
}
